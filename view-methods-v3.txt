 focus(actor,target,hero=false){this.focused=!!actor;this.focusActor=actor;this.focusTarget=target;const light=this.stage.querySelector('#battle-light');if(light){light.style.opacity=actor&&this.cinematic?(hero?'.48':'.24'):'0';light.style.background=actor?`radial-gradient(ellipse at ${actor.pos.x/W*100}% ${actor.pos.y/H*100}%,transparent 5%,#080b20 85%)`:'transparent'}for(const p of this.fighters)p.el?.classList.toggle('in-background',!!actor&&p!==actor&&p!==target);}
 mouth(p,target){const dx=target.pos.x-p.pos.x,dy=target.pos.y-p.pos.y,n=Math.max(1,Math.hypot(dx,dy));return {x:p.pos.x+dx/n*14,y:p.pos.y+dy/n*14-8};}
 scar(pos,type){const c=this.bg;c.save();c.globalAlpha=.25;c.fillStyle=type==='fire'?'#351f24':'#282638';c.beginPath();c.ellipse(pos.x,pos.y+27,23,9,0,0,Math.PI*2);c.fill();if(type==='ground'||type==='rock'){c.strokeStyle='#1c1826';c.lineWidth=2;for(let i=0;i<5;i++){const a=i*Math.PI*.4;c.beginPath();c.moveTo(pos.x,pos.y+22);c.lineTo(pos.x+Math.cos(a)*15,pos.y+22+Math.sin(a)*12);c.lineTo(pos.x+Math.cos(a+.2)*30,pos.y+22+Math.sin(a+.2)*18);c.stroke()}}c.restore();}
